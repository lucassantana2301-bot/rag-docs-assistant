90def _pct(value: float) -> str:

94def _chips(items: Iterable[str], css: str = "") -> str: 

95 return "".join(f'<span class="chip {css}">{escape(item)}</span>' for item in items)

98def _case_card(index: int, case: CaseResult) -> str: 

99 status = "pass" if case.passed else "fail" 

100 kind = "answerable" if case.answerable else "refusal" 

101 label = "passou" if case.passed else "falhou" 

102 if case.answerable: 

103 rank = ( 

104 f"fonte certa na posição {case.first_relevant_rank}" 

105 if case.first_relevant_rank 

106 else "fonte certa fora do top-k" 

107 ) 

108 checks = ( 

109 f"{rank} · cita a fonte esperada: {'sim' if case.cites_expected else 'não'}" 

110 f" · contém os fatos-chave: {'sim' if case.contains_facts else 'não'}" 

111 ) 

112 else: 

113 checks = "fora do escopo dos documentos · " + ( 

114 "recusou corretamente" if not case.grounded else "respondeu sem base" 

115 ) 

116 

117 citations = "".join( 

118 f'<span class="chip {"hit" if c.expected else ""}">[{c.number}] {escape(c.source)}' 

119 f"{' › ' + escape(c.heading) if c.heading else ''}</span>" 

120 for c in case.citations 

121 ) 

122 rows = "".join( 

123 "<tr>" 

124 f'<td class="num">{position}</td>' 

125 f'<td>{"✅ " if hit.relevant else ""}{escape(hit.source)}<div class="small">{escape(hit.heading)}</div></td>' 

126 f'<td class="num">{hit.lexical_rank or "—"}</td>' 

127 f'<td class="num">{hit.vector_rank or "—"}</td>' 

128 f'<td class="num">{hit.score:.4f}</td>' 

129 "</tr>" 

130 for position, hit in enumerate(case.retrieved, start=1) 

131 ) 

132 expected = ( 

133 f'<div class="small">Fontes esperadas</div><div class="chips">{_chips(case.expected_sources)}</div>' 

134 if case.expected_sources 

135 else "" 

136 ) 

137 facts = ( 

138 f'<div class="small">Fatos-chave exigidos</div><div class="chips">' 

139 f"{_chips(case.must_contain, 'hit' if case.contains_facts else 'miss')}</div>"

164def render_html( 

165 retrieval: list[RetrievalMetrics], 

166 answers: AnswerMetrics, 

167 cases: list[CaseResult], 

168 *, 

169 generator: str, 

170 generated_at: str, 

171 revision: str | None = None, 

172 repository_url: str | None = None, 

173) -> str: 

174 passed = sum(1 for case in cases if case.passed) 

175 hybrid = next((m for m in retrieval if m.mode == "hybrid"), retrieval[-1]) 

176 kpis = [ 

177 ("Perguntas aprovadas", f"{passed}/{len(cases)}"), 

178 (f"Hit rate@{hybrid.k} (híbrida)", _pct(hybrid.hit_rate)), 

179 ("MRR (híbrida)", f"{hybrid.mrr:.3f}"), 

180 ("Citam a fonte certa", _pct(answers.faithful_citation_rate)), 

181 ("Recusas corretas", _pct(answers.correct_refusal_rate)), 

182 ] 

183 kpi_html = "".join(f'<div class="kpi"><span>{escape(k)}</span><strong>{escape(v)}</strong></div>' for k, v in kpis) 

184 retrieval_rows = "".join( 

185 f"<tr><td>{escape(_MODE_LABELS.get(m.mode, m.mode))}</td><td class='num'>{_pct(m.hit_rate)}</td>" 

186 f"<td class='num'>{m.mrr:.3f}</td><td class='num'>{m.cases}</td></tr>" 

187 for m in retrieval 

188 ) 

189 cards = "".join(_case_card(i, case) for i, case in enumerate(cases, start=1)) 

190 source_link = f' · <a href="{escape(repository_url)}">código-fonte</a>' if repository_url else "" 

191 rev = f" · revisão <code>{escape(revision[:7])}</code>" if revision else "" 

192 return f"""<!doctype html> 

193<html lang="pt-BR"> 

194<head> 

195<meta charset="utf-8"> 

196<meta name="viewport" content="width=device-width, initial-scale=1">

199<style>{_CSS}</style> 

200</head> 

201<body> 

202<main> 

203 <h1>RAG Docs Assistant · relatório de avaliação</h1> 

204 <p class="lead">Respostas reais do pipeline (ingestão → busca híbrida → geração com citações) para as 

205 {len(cases)} perguntas do golden set, geradas no CI a cada push na <code>main</code>. 

206 Gerador: <code>{escape(generator)}</code>{rev}{source_link}.</p> 

207 

208 <div class="grid">{kpi_html}</div> 

209 

210 <h2>Recuperação por modo de busca</h2> 

211 <table> 

212 <thead><tr><th>Modo</th><th>Hit rate@{hybrid.k}</th><th>MRR</th><th>Perguntas</th></tr></thead>
